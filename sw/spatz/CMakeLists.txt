# K9R5: programs for the Spatz cluster (sw/spatz/*.c), built with Spatz's
# runtime (snRuntime, Cheshire platform) and LLVM toolchain. Configured and
# built by `make k9r5-sw-spatz`, which passes the cluster parameters from
# hw/spatz/k9r5.spatz_cluster.hjson; the build stays out of the spatz
# submodule.

cmake_minimum_required(VERSION 3.13)

set(SPATZ_SW_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../spatz/sw)
list(APPEND CMAKE_MODULE_PATH ${SPATZ_SW_DIR}/cmake)
set(CMAKE_TOOLCHAIN_FILE toolchain-llvm)
set(SNITCH_RUNTIME snRuntime-cluster CACHE STRING "")
# encoding.h from riscv-opcodes at Spatz's pin (generated by the Makefile)
set(K9R5_ENCODING_DIR "" CACHE PATH "Directory with encoding.h")
# Address of the results left for CVA6 (include/k9r5_spatz_mailbox.h)
set(K9R5_SPATZ_MAILBOX "0" CACHE STRING "Mailbox address in the cluster's DRAM window")

project(k9r5_spatz LANGUAGES C ASM)
include(SnitchUtilities)

include_directories(${K9R5_ENCODING_DIR})
# Spatz's Cheshire putchar.c passes integers as pointers, an error since
# clang 16; encoding.h defines a few Snitch CSRs twice
add_compile_options("$<$<COMPILE_LANGUAGE:C>:-Wno-error=int-conversion>" -Wno-macro-redefined)
add_subdirectory(${SPATZ_SW_DIR}/snRuntime snRuntime)

include_directories(${SNRUNTIME_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_compile_definitions(K9R5_SPATZ_MAILBOX=${K9R5_SPATZ_MAILBOX})
add_compile_options(-O3 -g)
# The programs contain hand-written RVV inline assembly (see toolchain-llvm)
add_compile_options("$<$<COMPILE_LANGUAGE:C>:-march=${SPATZ_MARCH_VECTOR}>")

file(GLOB K9R5_SPATZ_PROGS ${CMAKE_CURRENT_SOURCE_DIR}/*.c)
foreach(src ${K9R5_SPATZ_PROGS})
  get_filename_component(name ${src} NAME_WE)
  add_snitch_executable(${name} ${src})
  set_target_properties(${name} PROPERTIES SUFFIX ".elf")
endforeach()
